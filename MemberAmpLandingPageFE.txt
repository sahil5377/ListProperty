"use client";
import AuthLayout from '@/app/auth/components/layout/AuthLayout';
import { bbIconMapping } from '@/app/my-mls-listing/bbIconMapping';
import Styles from '@/app/my-mls-listing/components/bb-amp-landing-page/BBAmpLandiPage.module.scss';
import { ALLOWEDTEMPLATETYPES } from '@/app/my-mls-listing/components/bb-amp-landing-page/constants';
import { iconMapping } from '@/app/my-mls-listing/iconMapping';
import { useMyMlsListingStore } from '@/app/my-mls-listing/store/index';
import FooterAnimationGif from '@/assets/footerAnimation.gif';
import Globe from '@/assets/icons/globe.svg';
import InformationIcon from '@/assets/icons/information-icon.svg';
import Sparkles from '@/assets/icons/sparkles.svg';
import ViewGrid from '@/assets/icons/view-grid.svg';
import LoadingGif from "@/assets/loading.gif";
import LocationMarkerBB from '@/assets/location-marker-bb.svg';
import OpenHouseGif from '@/assets/openHouseGIF.gif';
import { Footer, NavBar } from '@/components';
import { useCaslContext } from '@/components/CaslContextAPI/CaslContext';
import { useWaitForImagesLoaded } from '@/hooks/useWaitForImagesLoaded';
import SafeTokenService from '@/services/modules/SafeToken.module';
import { useUserStore } from '@/store/commons/user';
import { generateEditorToolSession } from '@/utils';
import { CardMedia, createTheme, ThemeProvider } from '@mui/material';
import Image from 'next/image';
import Link from 'next/link';
import { useEffect, useRef, useState } from 'react';

const theme = createTheme({
  components: {
    MuiTableCell: {
      styleOverrides: {
        head: {
          '& .Mui-TableHeadCell-Content-Wrapper': {
            fontFamily: 'ppnns-bs',
            textTransform: 'uppercase',
          },
        },
        root: {
          '&.MuiTableCell-body': {
            paddingTop: '22px',
            paddingBottom: '22px',
            fontSize: '16px',
            fontFamily: 'popin-gr',
          },
        },
      },
    },
    MuiPopover: {
      styleOverrides: {
        paper: {
          border: '1px solid #58595A',
          borderRadius: '6px',
          boxShadow: '0px 2px 8px #c4c4c4',
        },
      },
    },
    MuiMenu: {
      styleOverrides: {
        paper: {
          borderRadius: '6px',
          border: '1px solid #58595A',
          boxShadow: '0px 2px 8px #c4c4c4',
          ul: {
            backgroundColor: 'white',
            li: {
              ':hover': {
                backgroundColor: '#EBEBEB',
              }
            }
          }
        },
        list: {
          padding: 0,
        },
      },
    },
  },
});

const MemberAmpLandingPage = () => {

  const getToken = useMyMlsListingStore((state) => state.getToken);
  const userLogged = useUserStore((state) => state.userLogged);
  const listingData = useMyMlsListingStore((state) => state.listingData);
  const justListedTemplates = useMyMlsListingStore((state) => state.justListedTemplates);
  const openHouseTemplates = useMyMlsListingStore((state) => state.openHouseTemplates);

  const [currentTemplateName, setCurrentTemplateName] = useState<string>('');
  const [currentSection, setCurrentSection] = useState<string>('Just Listed');
  const [currentSideMenuOptions, setCurrentSideMenuOptions] = useState<string[]>([]);
  const [templateOrder, setTemplateOrder] = useState<string[]>([]);
  const templateSectionRef = useRef<HTMLDivElement>(null);

  const safeToken = SafeTokenService();
  const { ability } = useCaslContext();
  const isAthenaEnabled = ability.can('view', 'EnableAthena');

  let imageUrls: string[] = Object.values(justListedTemplates).flat().map((template) => template.fileName);
  const imageUrlsOpenHouse: string[] = Object.values(openHouseTemplates).flat().map((template) => template.fileName);
  imageUrls = imageUrls.concat(imageUrlsOpenHouse);
  const { isLoading, loadedCount, total } = useWaitForImagesLoaded(imageUrls);

  // Initialize template order and side menu
  useEffect(() => {
    const templates = currentSection === 'Just Listed' ? justListedTemplates : openHouseTemplates;
    const filteredTemplateNames = Object.keys(templates).filter((name) =>
      ALLOWEDTEMPLATETYPES.includes(name)
    );
    setCurrentSideMenuOptions(filteredTemplateNames);
    setTemplateOrder(filteredTemplateNames);
    setCurrentTemplateName(filteredTemplateNames[0] || '');
  }, [currentSection, justListedTemplates, openHouseTemplates]);

  // Get token on user login
  useEffect(() => {
    if (userLogged.id) {
      getToken();
    }
  }, [userLogged.id]);

  const goEditorWithSafeToken = async (prettyUrl: string, brochureId: string) => {
    const uri = localStorage.getItem('token') || sessionStorage.getItem('token');
    const tenantUid = localStorage.getItem('tenant_uid')
    const response = await safeToken.getSafeToken(tenantUid, uri);

    await generateEditorToolSession(response.data.access_token);
    const userId = localStorage.getItem('userId') || sessionStorage.getItem('userId');
    let url = userId == null
      ? `${window.location.origin}/brochure/?tag=${prettyUrl}&bid=${brochureId}`
      : `${window.location.origin}/brochure/?tag=${prettyUrl}&bid=${brochureId}&user_id=${userId}`;

    if (isAthenaEnabled)
      url = userId == null
        ? `${window.location.origin}/athena?token=${response.data.access_token}&tag=${prettyUrl}&bid=${brochureId}`
        : `${window.location.origin}/athena?token=${response.data.access_token}&tag=${prettyUrl}&bid=${brochureId}&user_id=${userId}`;

    window.open(url, '_blank');
  };

  const handleSectionButton = (sectionName: string) => {
    setCurrentSection(sectionName);
    if (templateSectionRef.current) {
      const offsetTop = templateSectionRef.current.offsetTop;
      window.scrollTo({
        top: offsetTop - 150,
        behavior: 'smooth',
      });
    }
  };

  const handleTemplateTypeClick = (templateName: string) => {
    setCurrentTemplateName(templateName);
    setTemplateOrder((prevOrder) => [
      templateName,
      ...prevOrder.filter((name) => name !== templateName),
    ]);
  };

  return (
    <AuthLayout>
      <NavBar />
      <ThemeProvider theme={theme}>
        {isLoading && <div className={Styles.spinner}>
          <Image className={Styles['loading-item']} src={LoadingGif} alt="loading" />
          <p>{loadedCount}/{total}</p>
        </div>}
        <hr className={Styles.line} />
        <div>
          <div className={Styles['main-container']}>
            <div className={Styles['top-information-data']}>
              <div className={Styles['welcome']}>
                <div className={Styles['welcome-container']}>
                  <h2>Congrats on your listing<br />{`${userLogged?.firstName}`}!<br />Now let’s get marketing.</h2>
                  <p className={Styles['description']}>Below you will find everything you need to market your listing and grow your business.</p>
                  <div className={Styles['buttons']}>
                    <button
                      className={Styles['btn-open-house']}
                      onClick={() => handleSectionButton('Open House')}
                    >
                      Open House
                    </button>
                    <button
                      className={Styles['btn-just-listed']}
                      onClick={() => handleSectionButton('Just Listed')}
                    >
                      Just Listed
                    </button>
                  </div>
                  <div className={Styles['information-container']}>
                    <Image
                      src={InformationIcon}
                      alt="Information Icon"
                      className={Styles['information-icon']}
                    />
                    <Link href="#video-section">Remind me how to use my automated marketing package.</Link>
                  </div>
                </div>
              </div>
              <div className={Styles['listing-image']}>
                <div className={Styles['address']}>
                  <Image
                    src={LocationMarkerBB}
                    alt="Location Marker"
                    className={Styles['icon']}
                  />
                  <h3>{listingData?.fullStreetAddress}</h3>
                </div>
                <CardMedia
                  className={Styles['property-image']}
                  component="img"
                  image={listingData?.propertyImage}
                  alt="House"
                />
              </div>
            </div>
            <div className={Styles['open-house-and-listing']}>
              <img
                src={OpenHouseGif.src}
                alt="AMP Landing Page"
                className={Styles['styled-image']}
              />
              <div className={Styles['content-container']}>
                <div className={Styles['content']}>
                  <Image
                    src={Sparkles}
                    alt="Sparkles"
                    className={Styles['icon']}
                  />
                  <h5>Automatically<br />Customized for You</h5>
                  <p>All designs are already done for you, if you&lsquo;d like to edit you can or simply choose a delivery option and we will handle the rest.</p>
                </div>
                <div className={Styles['content']}>
                  <Image
                    src={ViewGrid}
                    alt="View Grid"
                    className={Styles['icon']}
                  />
                  <h5>Impactful Features</h5>
                  <p>A variety of features you can smile about: Customizable templates, print and ship to your mailing list, farm a new one with address mapping tool, text-to-yourself, your sphere or post on social.</p>
                </div>
                <div className={Styles['content']}>
                  <Image
                    src={Globe}
                    alt="Globe"
                    className={Styles['icon']}
                  />
                  <h5>Available to over<br />500 MLS&lsquo; Nationwide</h5>
                  <p>A powerful integration with ListHub provides robust listing data to automatically create marketing for you.</p>
                </div>
              </div>
            </div>
            <div ref={templateSectionRef} className={Styles['templates-main-container']}>
              <div className={Styles['left-menu']}>
                <h5 className={Styles['navigate-by']}>NAVIGATE BY</h5>
                <div className={Styles['buttons-section']}>
                  <button
                    className={`${Styles['button']} ${currentSection === 'Open House' ? Styles['active'] : Styles['inactive']
                      }`}

                    onClick={() => handleSectionButton('Open House')}
                  >
                    Open House
                  </button>
                  <button
                    className={`${Styles['button']} ${currentSection === 'Just Listed' ? Styles['active'] : Styles['inactive']
                      }`}

                    onClick={() => handleSectionButton('Just Listed')}
                  >
                    Just Listed
                  </button>
                </div>
                <div className={Styles['content-type']}>
                  {currentSideMenuOptions.length > 0 ? (
                    currentSideMenuOptions.map((templateName) => {
                      const cleanTitle = templateName.toLocaleLowerCase().replace(/[-\/]+/g, ' ').replace(/\s+/g, ' ').trim();
                      const templateIcon = bbIconMapping?.[cleanTitle] || iconMapping['one pager'];
                      return (
                        <div
                          key={templateName}
                          className={`${Styles['type']} ${currentTemplateName === templateName ? Styles['active'] : ''}`}
                          onClick={() => handleTemplateTypeClick(templateName)}
                        >
                          <Image src={templateIcon} alt="Icon type" className={Styles['icon-type']} />
                          <p className={Styles['title']}>{templateName}</p>
                        </div>
                      );
                    })
                  ) : (
                    <div className={Styles['type']}>
                      <h2 className={Styles['no-results']}>No data available</h2>
                    </div>
                  )}
                </div>
              </div>
              <div className={Styles['templates-section']}>
                {templateOrder.map((templateName) => {
                  if (!ALLOWEDTEMPLATETYPES.includes(templateName)) return null;
                  const templates = currentSection === 'Just Listed' ? justListedTemplates : openHouseTemplates;
                  const templateGroup = templates[templateName] || [];
                  let description = '';
                  if (templateName === 'Postcard') {
                    description = 'Direct Mail is the best way to market your items to be sent directly to your selected or uploaded list.';
                  } else if (templateName === 'Flyer') {
                    description = 'There are many options available below to market your new listing.';
                  } else if (templateName === 'Social Media') {
                    description = 'Easily share these to your social platforms by texting to self or directly to Facebook.';
                  }
                  return (
                    <div key={templateName}>
                      <div className={Styles['section-title']}>
                        <h5>{templateName}</h5>
                        {description && <p className={Styles['section-description']}>{description}</p>}
                      </div>
                      <div className={Styles['templates']}>
                        {templateGroup.length > 0 ? (
                          templateGroup.map((template) => (
                            <div
                              key={template.id}
                              className={Styles['template']}
                              onClick={() => goEditorWithSafeToken(template.templateName, template.brochureId)}
                            >
                              <div className={Styles['image-container']}>
                                <Image
                                  src={template.fileName}
                                  alt="Template image"
                                  className={Styles['template-image']}
                                  fill
                                />
                              </div>
                              <h6>{template.templateTitle}</h6>
                            </div>
                          ))
                        ) : (
                          <h2 className={Styles['no-results']}>No records to display</h2>
                        )}
                      </div>
                    </div>
                  );
                })}
              </div>
            </div>
            <div className={Styles['bottom-section']}>
              <div className={Styles['left-section']}>
                <h4>How does my <span>automated</span><br />marketing package work?</h4>
                <p>Within 24 hours of submitting a new listing to your MLS, you will receive an Automated Marketing Package via email that has 22 free, complete marketing templates pre-made for your listing. Learn more below.</p>
                <Link
                  className={Styles['amp-button']}
                  href='https://breakthroughbroker.lpages.co/automated-marketing-package/'
                  target='_blank'
                  rel='noopener noreferrer'
                >
                  Learn More
                </Link>
              </div>
              <div id="video-section" className={Styles['right-section']}>
                <iframe
                  src={'https://www.youtube.com/embed/L9cRc1OMiFg'}
                  frameBorder="0"
                  allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture"
                  allowFullScreen
                  title="Embedded youtube"
                />
              </div>
            </div>
            <div className={Styles['main-help-section']}>
              <div className={Styles['help-section']}>
                <h6>We want to help.</h6>
                <p>This a powerful tool and we don’t want you to miss out. Reach out and we will support you in every way we can to get you closer to saving time, saving money and selling more homes with Automated Marketing Packages. </p>
                <Link
                  className={`${Styles['amp-button']} ${Styles['link']}`}
                  href='/contact-us'
                  target='_blank'
                  rel='noopener noreferrer'
                >
                  Contact Support
                </Link>
              </div>
            </div>
            <div className={Styles['img-animation']}>
              <img
                src={FooterAnimationGif.src}
                alt="Marketing Animation"
              />
            </div>
          </div>
        </div>
        <Footer />
      </ThemeProvider>
    </AuthLayout >)
}
export default MemberAmpLandingPage;
