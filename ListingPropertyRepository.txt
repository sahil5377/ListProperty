using BBMC.API.MarketingCenter.Models.DTOs;
using BBMC.API.MarketingCenter.Repository.Contracts;
using BBMC.CoreService.Models.Constants;
using BBMC.CoreService.Models.Contracts;
using BBMC.CoreService.Models.Entities;
using MongoDB.Bson;
using MongoDB.Driver;
using System.Diagnostics.CodeAnalysis;

namespace BBMC.API.MarketingCenter.Repository.Services
{
    public class ListingPropertyRepository : IListingPropertyRepository
    {
        private readonly IMongoCollection<TblMlsListingProperty> _listingProperty;
        private readonly ILogger<ListingPropertyRepository> _logger;

        public ListingPropertyRepository(IMongoContext mongoClient, ILogger<ListingPropertyRepository> logger)
        {
            _listingProperty = mongoClient.Database.GetCollection<TblMlsListingProperty>(NoSqlCollectionNames.MlsListingProperties);
            _logger = logger;
        }

        public async Task<ListingPropertyDto> GetListingProperty(string listingKey, int userId)
        {
            try
            {
                var joinCollections = new List<BsonDocument>
                {
                    new BsonDocument("$match",
                        new BsonDocument
                        {
                            { "listing_key", listingKey }
                        }
                    ),
                    new BsonDocument("$lookup", new BsonDocument
                    {
                        { "from", "tbl_property_brochures" },
                        { "let", new BsonDocument
                            {
                                { "listing_key", "$listing_key" },
                                { "user_id", userId }
                            }
                        },
                        { "pipeline", new BsonArray
                            {
                                new BsonDocument("$match", new BsonDocument
                                {
                                    { "$expr", new BsonDocument("$and", new BsonArray
                                        {
                                            new BsonDocument("$eq", new BsonArray { "$$listing_key", "$listing_key" }),
                                            new BsonDocument("$eq", new BsonArray { "$$user_id", "$user_id" })
                                        })
                                    }
                                }),
                                new BsonDocument("$lookup", new BsonDocument
                                {
                                    { "from", "tbl_brochure" },
                                    { "localField", "brochure_id" },
                                    { "foreignField", "_id" },
                                    { "as", "brochure_info" }
                                }),
                                new BsonDocument("$unwind", "$brochure_info"),
                                new BsonDocument("$lookup", new BsonDocument
                                {
                                    { "from", "brochuretemplates" },
                                    { "localField", "brochure_info.template_name" },
                                    { "foreignField", "pretty_url" },
                                    { "as", "brochure_templates_info" }
                                }),
                                 new BsonDocument("$unwind", new BsonDocument
                                {
                                    { "path", "$brochure_templates_info" },
                                    { "preserveNullAndEmptyArrays", true }
                                }),
                                new BsonDocument("$project", new BsonDocument
                                {
                                    { "TemplateName", "$template_name" },
                                    { "TemplateTitle", "$brochure_templates_info.title" },
                                    { "BrochureId", "$brochure_id" },
                                    { "FileName", "$file_name" },
                                    { "HtmlUrl", "$html_url" },
                                    { "DigitalTemplate", "$digital_template" }
                                })
                            }
                        },
                        { "as", "property_brochures" }
                    }),
                    new BsonDocument("$project", new BsonDocument
                    {
                        { "ParticipantId", "$participant_id" },
                        { "FullStreetAddress", "$full_street_address" },
                        { "UserId", "$user_id"},
                        { "PropertyImage", "$property_image"},
                        { "PropertyBrochures", "$property_brochures"},
                    })
                };

                return await _listingProperty.Aggregate<ListingPropertyDto>(joinCollections).FirstAsync();
            }
            catch (Exception ex)
            {
                throw;
            }
        }
   }
}
