[HttpGet]
[ProducesResponseType(200, Type = typeof(ResponseDto<ListingPropertyDto>))]
[ProducesResponseType(500, Type = typeof(ResponseDto<ListingPropertyDto>))]
[Authorization(Role.AnyRole)]
public async Task<IActionResult> GetListingProperty([FromQuery] string ListingKey, [FromQuery] int userId)
{
    ResponseDto<ListingPropertyDto> response = new();
    try
    {
        response.Data = await _listingPropertyService.GetListingProperty(ListingKey, userId);
        _logger.LogInformation($"Get Listing Property, ListingKey:{ListingKey}", ModuleService.ListingProperties);
        return Ok(response);
    }
    catch (Exception ex)
    {
        _logger.LogError("ListingPropertyController GetListingProperty() Log => {@result}", ex);
        return StatusCode(StatusCodes.Status500InternalServerError, ResponseError(response, ex));
    }
    finally
    {
        if (!response.Success)
            _logger.LogError($"Error Get Listing Property, ListingKey: {ListingKey}", ModuleService.ListingProperties);
    }
}
